// 🔐 Firebase Config const firebaseConfig = { apiKey: "YOUR_API_KEY", authDomain: "YOUR_PROJECT.firebaseapp.com", projectId: "YOUR_PROJECT_ID", }; firebase.initializeApp(firebaseConfig); const auth = firebase.auth(); // 🚀 Main Class class RetroFutureLoginForm { constructor() { this.form = document.getElementById('loginForm'); this.emailInput = document.getElementById('email'); this.passwordInput = document.getElementById('password'); this.passwordToggle = document.getElementById('passwordToggle'); this.successMessage = document.getElementById('successMessage'); this.socialButtons = document.querySelectorAll('.social-retro'); this.init(); } init() { this.form.addEventListener('submit', (e) => this.handleSubmit(e)); this.passwordToggle.addEventListener('click', () => this.togglePassword()); this.socialButtons.forEach(btn => { btn.addEventListener('click', () => { const provider = btn.innerText.trim(); this.socialLogin(provider); }); }); } togglePassword() { this.passwordInput.type = this.passwordInput.type === "password" ? "text" : "password"; } // ✅ Email Login (Demo) handleSubmit(e) { e.preventDefault(); const email = this.emailInput.value; const password = this.passwordInput.value; if (!email || !password) { alert("Fill all fields"); return; } // Save user localStorage.setItem("user", JSON.stringify({ email })); this.showSuccess(); } // 🔥 Google + GitHub Login async socialLogin(provider) { try { let providerInstance; if (provider.includes("GOOGLE")) { providerInstance = new firebase.auth.GoogleAuthProvider(); } else if (provider.includes("GITHUB")) { providerInstance = new firebase.auth.GithubAuthProvider(); } const result = await auth.signInWithPopup(providerInstance); // Save user localStorage.setItem("user", JSON.stringify(result.user)); this.showSuccess(); } catch (error) { alert(error.message); } } // ✅ Success + Redirect showSuccess() { document.querySelector('.future-form').style.display = "none"; document.querySelector('.future-social').style.display = "none"; this.successMessage.classList.add("show"); setTimeout(() => { window.location.href = "dashboard.html"; }, 2000); } } // Init document.addEventListener("DOMContentLoaded", () => { new RetroFutureLoginForm(); });
